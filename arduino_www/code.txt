const int sensorPin = A0;

void setup()
{
    Serial.begin(9600);
}

void loop()
{
    const int samples = 20;
    long sum = 0;

    for (int i = 0; i < samples; i++)
    {
        sum += analogRead(sensorPin);
        delay(5);
    }

    float raw =
        (float)sum / samples;

    float voltage =
        raw * (5.0 / 1023.0);

    float temperature =
        voltage * 100.0;

    // ВАЖНО: выводим только число
    Serial.println(temperature, 1);

    delay(900);
}
